#include "gemm.h"

// CHANGE: Reorder loops to better utilize DSPs and reduce BRAM usage
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
#pragma HLS ARRAY_PARTITION variable=A complete=1
#pragma HLS ARRAY_PARTITION variable=B complete=1
#pragma HLS ARRAY_PARTITION variable=C complete=1
    for (int k = 0; k < N; k++) {
#pragma HLS UNROLL factor=2
        for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE II=1
            for (int j = 0; j < N; j++) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }
}
