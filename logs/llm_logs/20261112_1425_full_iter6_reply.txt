```cpp
#include "gemm.h"

// CHANGE: Introduce a temporary array for partial results and use pipelining
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
    acc_t temp[N][N] = {0};

    for (int i = 0; i < N; i++) {
        for (int k = 0; k < N; k++) {
#pragma HLS PIPELINE II=2
            for (int j = 0; j < N; j++) {
                temp[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            C[i][j] = temp[i][j];
        }
    }
}
```

This change introduces a temporary array `temp` to store intermediate results, which helps in reducing the number of memory accesses and improves the pipelining efficiency. The outer loop over `i` and `k` is pipelined with an initiation interval of 2, as required.
