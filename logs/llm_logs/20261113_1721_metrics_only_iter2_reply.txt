```cpp
#include "gemm.h"

// CHANGE: Unroll the inner loop
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
#pragma HLS PIPELINE II=1
            acc_t sum = 0;
            for (int k = 0; k < N; k++) {
                sum += A[i][k] * B[k][j];
            }
            C[i][j] = sum;
        }
    }
}
```

This change attempts to unroll the inner loop using a `#pragma HLS PIPELINE II=1` directive to improve the initiation interval (II). This should help reduce the latency while staying within the resource limits.
