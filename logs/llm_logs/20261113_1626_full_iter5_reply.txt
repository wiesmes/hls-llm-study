#include "gemm.h"

// CHANGE: Reduce the initiation interval by pipelining the outer loop and unrolling the inner loop
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
    for (int i = 0; i < N; i++) {
#pragma HLS loop_tripcount max=N min=N
#pragma HLS pipeline II=1
        for (int j = 0; j < N; j += 4) {
#pragma HLS loop_tripcount max=N/4 min=N/4
#pragma HLS unroll factor=4
            acc_t sum = 0;
            for (int k = 0; k < N; k++) {
                sum += A[i][k] * B[k][j];
                sum += A[i][k] * B[k][j+1];
                sum += A[i][k] * B[k][j+2];
                sum += A[i][k] * B[k][j+3];
            }
            C[i][j] = sum;
            C[i][j+1] = sum;
            C[i][j+2] = sum;
            C[i][j+3] = sum;
        }
    }
}
