#include "gemm.h"

// CHANGE: Optimize memory access patterns and use pipelining
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
    // Temporary array for partial results
    acc_t temp[N][N];

    // Loop to compute the partial results
    for (int i = 0; i < N; i++) {
        for (int k = 0; k < N; k++) {
#pragma HLS PIPELINE II=2
            for (int j = 0; j < N; j++) {
                temp[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    // Loop to finalize the results
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            C[i][j] = temp[i][j];
        }
    }
}
