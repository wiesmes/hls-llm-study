#include "gemm.h"

// CHANGE: Reorganize memory access patterns and optimize pipelining to reduce initiation interval
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
    for (int i = 0; i < N; i++) {
#pragma HLS loop_tripcount max=N min=N
        for (int j = 0; j < N; j++) {
#pragma HLS loop_tripcount max=N min=N
#pragma HLS pipeline II=1
            acc_t sum = 0;
            for (int k = 0; k < N; k += 8) {
#pragma HLS unroll factor=8
                sum += A[i][k] * B[k][j];
                sum += A[i][k+1] * B[k+1][j];
                sum += A[i][k+2] * B[k+2][j];
                sum += A[i][k+3] * B[k+3][j];
                sum += A[i][k+4] * B[k+4][j];
                sum += A[i][k+5] * B[k+5][j];
                sum += A[i][k+6] * B[k+6][j];
                sum += A[i][k+7] * B[k+7][j];
            }
            C[i][j] = sum;
        }
    }
}
