#include "gemm.h"

// CHANGE: Optimize memory access patterns and use pipelining
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
    acc_t sum[N];
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
#pragma HLS PIPELINE II=2
            sum[j] = 0;
            for (int k = 0; k < N; k++) {
                sum[j] += A[i][k] * B[k][j];
            }
        }
        C[i] = sum;
    }
}
