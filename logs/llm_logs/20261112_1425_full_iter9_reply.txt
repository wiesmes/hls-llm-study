#include "gemm.h"

// CHANGE: Introduce a temporary array for partial results and use pipelining
void gemm(data_t A[N][N], data_t B[N][N], acc_t C[N][N]) {
    acc_t C_temp[N][N];
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
#pragma HLS PIPELINE II=2
            acc_t sum = 0;
            for (int k = 0; k < N; k++) {
                sum += A[i][k] * B[k][j];
            }
            C_temp[i][j] = sum;
        }
    }
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            C[i][j] = C_temp[i][j];
        }
    }
}
